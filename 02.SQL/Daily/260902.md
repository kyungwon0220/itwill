### LENGTH(), LENGTHB()
```SQL
SELECT LENGTH('oracle'), LENGTHB('oracle'), LENGTH('오라클'), LENGTHB('오라클')
FROM DUAL;
```
- LENGTH(x) == x 문자열의 길이
- 한글은, 글자당 3BYTE

|LENGTH('oracle')|LENGTHB('oracle')||LENGTH('오라클')|LENGTHB('오라클')|
---:|---:|---|---:|---:|
6|6||3|9|

### TRIM(), LTRIM(), RTRIM()
```SQL
SELECT TRIM('a' FROM 'aaoracleaassaa')
FROM DUAL; /* 양쪽에 연속되는 ' a ' 제거 */

SELECT TRIM(BOTH 'a' FROM 'aaoracleaassaa')
FROM DUAL; /* == trim() */


SELECT LTRIM('aaoracleaassaa', 'a')
FROM DUAL; /* 왼쪽에 연속되는 'a' 제거 */

SELECT TRIM(LEADING 'a' FROM 'aaoracleaassaa')
FROM DUAL; /* == ltrim() */


SELECT RTRIM('aaoracleaassaa', 'a')
FROM DUAL; /* 오른쪽에 연속되는 'a' 제거 */

SELECT TRIM(TRAILING 'a' FROM 'aaoracleaassaa')
FROM DUAL; /* == rtrim() */
```
|TRIM|TRIM(BOTH *||LTRIM|TRIM(LEADING *||RTRIM|TRIM(TRAILING *|
:---|:---|---|:---|:---|---|:---|:---|
oracleaass|oracleaass||oracleaassaa|oracleaassaa||aaoracleaass|aaoracleaass|
---

### REPLACE()
```SQL
SELECT REPLACE('100-001', '-', '%')
FROM DUAL; /* 치환 */

SELECT REPLACE('-1-00-001-', '-', '')
FROM DUAL; /* 특정 문자 제거 가능 */

SELECT REPLACE('     100     001     ', ' ', '')
FROM DUAL; /* 공백 제거 가능 */
```
|REPLACE('100-001', '-', '%')|REPLACE('-1-00-001-', '-', '')|REPLACE('     100     001     ', ' ', '')|
:---|:---|:---|
100%001|100001||100001|

### LPAD(), RPAD()
```SQL
--[문제16] salary에 있는 값을 1000당*출력해주세요.
--SALARY STAR
------- -----
--5000 *****
--2000 **
--SELECT employee_id, salary, LPAD( ' ', salary/1000 + 1, '*' )
SELECT employee_id, salary, RPAD( ' ', salary/1000 + 1, '*' )
FROM hr.employees;
```
- LPAD == 원본 문자열을 기준으로, 왼쪽에 채움
- RPAD == 원본 문자열을 기준으로, 오른쪽에 채움
- 인자 == ( 원본 문자열, 얼마나 채울 것인지의 총 길이, 채울 문자 )
- 첫 인자로, ' ' 공백조차 넣지 않는다면, '' == NULL 이므로 결과값으로 NULL 출력

employee_id|salary|LPAD|RPAD|
:---|---:|:---|:---|
206|8300|********() | ()********|
205|12008|************() | ()************|
204|10000|**********() | ()**********|
. . .|. . .|. . .|. . .|
> 실제로 ' () ' 괄호가 출력되지는 않고, 원본 문자열의 공백 위치를 표시

### ROUND(), TRUNC(), CEIL(), MOD(), POWER(), ABS(), SQRT()
```SQL
SELECT ROUND(55.926, -2), ROUND(55.926, 2)
FROM DUAL; /* 지정한 자릿수에서 반올림 */

SELECT TRUNC(155.926, -2), TRUNC(155.926, 2)
FROM DUAL; /* 지정한 자릿수에서 버림 */

SELECT CEIL(10.001)
FROM DUAL; /* 올림 */

SELECT MOD(12, 7)
FROM DUAL; /* 나머지 값을 반환 */

SELECT 2*2*2, POWER(2, 3)
FROM DUAL; /* 제곱승 */

SELECT ABS(-15) "Absoute"
FROM DUAL; /* 절대값 */

SELECT SQRT(9) "Square root"
FROM DUAL; /* 루트 적용값(제곱근 == Square root) */
```

|ROUND(55.926, -2)|ROUND(55.926, 2)||TRUNC(155.926, -2)|TRUNC(155.926, 2)||CEIL|MOD|POWER|ABS|SQRT|
:---|:---|---|:---|:---|---|:---|:---|:---|:---|:---|
100|55.93||100|155.92||11|5|8|15|3|
