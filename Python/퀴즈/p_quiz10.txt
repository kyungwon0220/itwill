09. 만능 수치 제어(올림/내림/반올림) 함수에 관련된 코드에서 빈칸을 채우시오. 

import math

def custom_numeric_control(value, n, mode="floor"):
    """
    value: 원본 숫자 (실수)
    n: 최종적으로 남기고 싶은 소수점 자릿수 (양수: 소수점 아래, 음수: 십/백의 자리)
    mode: 'ceil'(올림), 'floor'(내림), 'round'(반올림)
    """
    # [원리] 10의 n승을 곱해 소수점을 이동시킨 후 함수를 적용하고, 다시 나누어 원상복구
    # 예: n이 2이면 factor는 100이 됩니다.
    factor = [ 빈칸 A ]
    
    if mode == "ceil":
        return math.ceil(value * factor) / factor
        
    elif mode == "floor":
        return [ 빈칸 B ]
        
    elif mode == "round":
        # 일관성 있는 자릿수 제어를 위해 다른 모드와 동일한 시프팅 메커니즘을 적용합니다.
        return [ 빈칸 C ]
        
    else:
        return value

# --- 실행 및 검증부 ---
test_num = 1234.56789

print(f"-> 소수점 3째자리까지 올림: {custom_numeric_control(test_num, 3, 'ceil')}")   
# 출력: 1234.568
print(f"-> 소수점 1째자리까지 내림: {custom_numeric_control(test_num, 1, 'floor')}")  
# 출력: 1234.5
