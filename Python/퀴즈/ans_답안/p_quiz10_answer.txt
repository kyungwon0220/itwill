import math

def custom_numeric_control(value, n, mode="floor"):
    """
    value: 원본 숫자
    n: 남기고 싶은 소수점 자릿수 (양수: 소수점 아래, 음수: 십의자리/백의자리 등)
    mode: 'ceil'(올림), 'floor'(내림), 'round'(반올림)
    """
    # 10의 n승 계산 (예: 2자리 남기려면 100, 3자리 남기려면 1000)
    factor = 10 ** n
    
    if mode == "ceil":
        return math.ceil(value * factor) / factor
    elif mode == "floor":
        return math.floor(value * factor) / factor
    elif mode == "round":
        return round(value, n-1)
    else:
        return value

import random

def play_rock_paper_scissors():
   
    print('가위 바위 보 게임에 오신 것을 환영합니다!')
    choices = ['가위', '바위', '보']
    computer_choice = random.choice(choices)

    user_choice = input("가위, 바위, 보 중 하나를 선택하세요: ").strip()
    if user_choice in choices:    
        print(f'당신의 선택: {user_choice}')
        print(f'컴퓨터의 선택: {computer_choice}')
        # 무승부!, 당신이 이겼습니다!, 컴퓨터가 이겼습니다!
        if user_choice == computer_choice :
            print('무승부!')
        # 사용자가 이기는 경우를 논리합 이용해서 조건 명령문 실행 
        elif (user_choice == '가위' and computer_choice == '보') or \
            (user_choice == '바위' and computer_choice == '가위') or \
            (user_choice == '보' and computer_choice == '바위'):
            print("당신이 이겼습니다!")
        else:
            print('컴퓨터가 이겼습니다!')
    else:
       print('잘못된 입력입니다. ')
       
    print('게임을 종료합니다')

# 파일명: student_logic.py

def analyze_student_data(student_list):
    results = []
    
    for s in student_list:
        scores = s["scores"]
        
        # 1. 총점 및 평균 계산
        total = sum(scores)
        avg = total / len(scores)
        
        # 2. 합격 여부 판별 (for 반복문 직접 활용)
        is_pass = True
        for score in scores:
            if score < 60:
                is_pass = False
                break
               
        # 3. 결과 저장 (if-else문 풀어서 작성)
        if is_pass:
            status = "PASS"
        else:
            status = "FAIL"
            
        results.append({
            "name": s["name"],
            "avg": avg,
            "status": status
        })
        
    return results

    best_student = student_list_result[0]  # 첫 번째 학생을 일단 최고로 가정

        if student_list_result[i]["avg"] > best_student["avg"]:
            best_student = student_list_result[i]
    return best_student
