- DECODE, NVL2() 및 MAX(), MIN() 등의 그룹 함수같은 SQL 전용 함수들은,  PL/SQL 문법 영역에서 직접 사용 불가 ( SQL 문장 안에서는, 사용 가능 )
  - ```SQL
    BEGIN
      --res := DECODE(x, 'Y', '사용', '미사용'); /* PL/SQL 문법 영역에서, SQL 전용 함수를 직접 사용 불가 ERR */


      DBMS_OUTPUT.PUT_LINE(UPPER('xyz')); /* ' XYZ ' 결과 정상 출력 ( UPPER() 사용 가능 )*/
    END;
    /
    ```
  - ```SQL
    DECLARE
      pct NUMBER; /* 기본값 == NULL */
      total NUMBER;
    BEGIN
      total := (100 * NVL(pct,1)); /* NVL() 사용 가능 */
      DBMS_OUTPUT.PUT_LINE(total);


      --    total := 100 * NVL2(pct,1, 2); /* NVL2() 사용 불가 ERR */


      DBMS_OUTPUT.PUT_LINE(TO_DATE('2026-08-19', 'yyyy-mm-dd')); /* TO_CHAR() 사용 가능 */

    
      DBMS_OUTPUT.PUT_LINE(NEXT_DAY(SYSDATE, '월요일')); /* NEXT_DAY() 사용 가능 */
    END;
    /
    ```
