-  화씨를 섭씨로 변환하는 공식
섭씨온도 : (화씨온도 - 32) * 5.0 / 9.0  
- try~ except ~ else ~ finally ~를 이용하여 화씨온도의 데이타 오류시 예외처리 필요
   유효한 숫자가 아닙니다. 숫자를 입력하세요.

# 결과2 :

**정답**
def fahrenheit_to_celsius():
    try:
        fahrenheit = float(input("화씨 온도를 입력하세요: "))  # 사용자 입력 받기
        celsius = (fahrenheit - 32) * 5.0 / 9.0  # 화씨를 섭씨로 변환
    except ValueError:
        print("유효한 숫자가 아닙니다. 숫자를 입력하세요.")
    else:
        print(f"{fahrenheit} 화씨는 {celsius:.2f} 섭씨입니다.")
    finally:
        print("프로그램이 종료되었습니다.")


fahrenheit_to_celsius()

import random

def number_guessing_game():
    target_number = random.randint(1, 100)  # 1~100 사이 랜덤 숫자 생성
    attempts = 0  # 시도 횟수

    print("1부터 100 사이의 숫자를 맞춰보세요!")

    while True:
        try:
            guess = int(input("숫자를 입력하세요: "))
            attempts += 1  # 시도 횟수 증가

            if guess < target_number:
                print("\n더 큰 숫자를 입력하세요!")
            elif guess > target_number:
                print("\n더 작은 숫자를 입력하세요!")
            else:
                print(f"\n축하합니다! {attempts}번 만에 정답({target_number})을 맞췄습니다!")
                break  # 게임 종료
        except ValueError:
            print("\n올바른 숫자를 입력하세요!")

# 게임 실행
number_guessing_game()

import random

def make_name(url, op, first_name, last_name1, last_name2):
    name_list = []
    while True:
        x = random.choice(first_name)
        y = random.choice(last_name1)
        z = random.choice(last_name2)
        if len(name_list) >= 7:
            break
        if (x + y + z) not in name_list:
            name_list.append(x + y + z)
    name_list.sort()
    print(name_list)
    with open(url, 'w', encoding=op) as file:
        for name in name_list:
            file.write(name + '\n')
        print(f' {url} : 파일 쓰기가 완료되었습니다...')
