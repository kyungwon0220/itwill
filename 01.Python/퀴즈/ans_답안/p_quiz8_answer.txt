9. BMI(체질량지수)를 계산하고, 결과에 따라 메시지를 출력하는 코드를 작성하시오. 

# BMI 계산: 체중(kg) / (신장(m) ** 2)
# BMI 지수에 따른 메세지 
bmi < 18.5 => 저체중
18.5 <= bmi < 24.9 => 정상 체중
25 <= bmi < 29.9 => 과체중
bmi >= 29.9 => 비만 


# 함수 호출 사용 예시
weight = 70  # 체중 (kg)
height = 1.75  # 신장 (m)
calculate_bmi(weight, height)


# 결과 예시 
BMI: 22.86, 상태: 정상 체중



 # 정답 
def calculate_bmi(weight, height):
    # BMI 계산: 체중(kg) / (신장(m) ** 2)
    bmi = weight / (height ** 2)

    # BMI에 따른 메시지 출력
    if bmi < 18.5:
        message = "저체중"
    elif 18.5 <= bmi < 24.9:
        message = "정상 체중"
    elif 25 <= bmi < 29.9:
        message = "과체중"
    else:
        message = "비만"
    
    # 결과 출력
    print(f"BMI: {bmi:.2f}, 상태: {message}")


weight = 70  # 체중 (kg)
height = 1.75  # 신장 (m)
calculate_bmi(weight, height)



10. 특정 년도를 함수로 전달하면 윤년인지 평년인지 출력하는 함수를 정의하고 호출하여라 

윤년 규칙 
- 연도가 4로 나누어 떨어지면 윤년.
- 그러나 연도가 100으로 나누어 떨어지면 윤년이 아니다.
- 다만 연도가 400으로 나누어 떨어지면 윤년.


# print() 문안에 함수 호출 예시 
print(f' 2000 년도는 {is_leap_year(2000)}')
print(f' 2025 년도는 {is_leap_year(2000)}')
print(f' 2030 년도는 {is_leap_year(2000)}')

# 결과 예시 
 1988 년도는 윤년
 2000 년도는 윤년
 2025 년도는 평년
 2030 년도는 평년



# 정답 
def is_leap_year(year):
    if (year % 4 == 0 and year % 100 != 0) or (year % 400 == 0):
        return '윤년'
    else:
        return '평년'
    
print(f' 1988 년도는 {is_leap_year(1988)}')
print(f' 2000 년도는 {is_leap_year(2000)}')
print(f' 2025 년도는 {is_leap_year(2025)}')
print(f' 2030 년도는 {is_leap_year(2030)}')



11. 학생명, 국어, 영어, 수학, 과학 점수를 함수로 전달하면 총점, 평균, 학점이 나오도록 
프로그래밍하시오. 

학점 기준:
90점 이상 → A
80점 이상 → B
70점 이상 → C
60점 이상 → D
60점 미만 → F

# 함수 호출 예시
calculate_student_score("홍길동", 85, 90, 78, 92)
calculate_student_score("김철수", 55, 60, 58, 62)
calculate_student_score("이영희", 95, 92, 88, 94)


# 결과 예시 
학생명: 홍길동
총점: 345, 평균: 86.25, 학점: B

학생명: 김철수
총점: 235, 평균: 58.75, 학점: F

학생명: 이영희
총점: 369, 평균: 92.25, 학점: A



# 정답 
def calculate_student_score(name, korean, english, math, science):
    total = korean + english + math + science  # 총점 계산
    average = total / 4  # 평균 계산
    
    # 학점 계산
    if average >= 90:
        grade = "A"
    elif average >= 80:
        grade = "B"
    elif average >= 70:
        grade = "C"
    elif average >= 60:
        grade = "D"
    else:
        grade = "F"
    
    # 결과 출력
    print(f"학생명: {name}")
    print(f"총점: {total}, 평균: {average:.2f}, 학점: {grade}")

calculate_student_score("홍길동", 85, 90, 78, 92)
calculate_student_score("김철수", 55, 60, 58, 62)
calculate_student_score("이영희", 95, 92, 88, 94)

# 학점 기준

# 정답

stGradeList = [['김수현', 85, 78, 88],
               ['이준호', 90, 85, 95],
               ['박지민', 70, 60, 75],
               ['최예린', 95, 98, 92],
               ['정하윤', 88, 80, 85]]


def calculate_grades(stGradeList):
    print("학생 성적표")
    print("=" * 40)

    for student in stGradeList:
        name = student[0]  # 학생 이름
        scores = student[1:]  # 점수 리스트
        avg = sum(scores) / len(scores)  # 평균 계산
        
        # 학점 계산 (함수 내부에 직접 구현)
        if avg >= 90:
            grade = "A"
        elif avg >= 80:
            grade = "B"
        elif avg >= 70:
            grade = "C"
        elif avg >= 60:
            grade = "D"
        else:
            grade = "F"
        
        print(f"이름: {name}, 평균: {avg:.2f}, 학점: {grade}")


calculate_grades(stGradeList)

 # 도서 데이터를 저장할 딕셔너리
library = {}

def add_book(book_id, title, author, year):
    """새로운 도서를 추가하는 함수"""
    if book_id in library:
        print(" 이미 존재하는 도서입니다!")
    else:
        library[book_id] = {"제목": title, "저자": author, "출판년도": year}
        print(f" '{title}' 도서가 추가되었습니다.")


def list_books(library):
    """전체 도서 목록을 출력하는 함수"""
    if not library:
        print("\n 도서 목록이 비어 있습니다.\n")
    else:
        print("\n 도서 목록:")
        for book_id, info in library.items():
            print(f" {book_id}: {info['제목']} - {info['저자']} ({info['출판년도']}) ")



list_books(library)

add_book(1, "파이썬 프로그래밍", "홍길동", 2023)
add_book(2, "데이터 사이언스 입문", "이몽룡", 2022)
