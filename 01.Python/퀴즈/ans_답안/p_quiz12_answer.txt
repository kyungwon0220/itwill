'''


# 프로그래밍 예시

import csv

# 1. 샘플 TSV 파일 생성 (10개 데이터)
file_path = 'countries.tsv'
data_to_save = [
    ['Country', 'Capital', 'Language'],
    ['Vietnam', 'Hanoi', 'Vietnamese'],
    ['Germany', 'Berlin', 'German'],
    ['Brazil', 'Brasilia', 'Portuguese'],
    ['Canada', 'Ottawa', 'English/French'],
    ['Japan', 'Tokyo', 'Japanese'],
    ['Australia', 'Canberra', 'English'],
    ['France', 'Paris', 'French'],
    ['Italy', 'Rome', 'Italian'],
    ['Spain', 'Madrid', 'Spanish'],
    ['Korea', 'Seoul', 'Korean']
]

with open(file_path, 'w', encoding='utf-8', newline='') as f:
    writer = csv.writer(f, delimiter='\t')
    writer.writerows(data_to_save)

print(f"--- {file_path} 파일이 생성되었습니다. ---\n")

# 2. TSV 파일을 읽어 출력
with open(file_path, 'r', encoding='utf-8') as f:
    # DictReader를 사용해 딕셔너리 형태로 읽기
    reader = csv.DictReader(f, delimiter='\t')
    data_list = list(reader)

    print(f"{'국가':<15} | {'수도':<12} | {'사용 언어'}")
    print("-" * 45)
    for row in data_list:
        print(f"{row['Country']:<15} | {row['Capital']:<12} | {row['Language']}")


# 3. 영어, 프랑스, 독일어를 사용하는 국가 목록을 출력하여라
print("\n[영어, 프랑스, 독일어를 사용하는 국가 목록]")
speaking_countries = [
    row for row in data_list 
    if 'English' in row['Language'] or 'French' in row['Language'] or 'German' in row['Language']
]

for row in speaking_countries:
    print(f"{row['Country']}", end=" ")
print()
