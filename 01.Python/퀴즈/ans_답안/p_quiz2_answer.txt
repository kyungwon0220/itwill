# 14. 다음 코드의 실행 결과는 무엇인가요?

text = "  Hello, World! Python is awesome!  "
result = text.strip().replace("World", "Universe").lower().split()
print(result)


	a. ['hello', 'universe', 'python', 'is', 'awesome']
	b. ['Hello', 'Universe', 'Python', 'is', 'awesome']
	c. ['hello,', 'universe!', 'python', 'is', 'awesome!']
	d. ['hello', 'world', 'python', 'is', 'awesome']


# c
# strip()은 양쪽 공백을 제거하고, replace()는 World를 Universe로, 
# lower()는 모두 소문자로, split()은 공백을 기준으로 나누어 리스트로 반환


# 15. 사용자로부터 날짜를 "YYYY-MM-DD" 형식으로 입력받고, 
해당 날짜의 연도, 월, 일만 아래와 같은 형태로  출력되도록 프로그래밍 하여라. 


날짜를 'YYYY-MM-DD' 형식으로 입력하세요: 2025-12-10
입력된 날짜: 2025년 12월 10일



# 사용자로부터 날짜 입력 받기
date = input("날짜를 'YYYY-MM-DD' 형식으로 입력하세요: ")

# 슬라이싱을 이용해 연, 월, 일 추출
year = date[:4]
month = date[5:7]
day = date[8:]

# 결과 출력
print(f"입력된 날짜: {year}년 {month}월 {day}일")
