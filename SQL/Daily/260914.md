### 제약 조건
#### PRIMARY KEY
- 테이블의 고유한 대표키로서, 1개
-  ' NULL ' 값 불가
-  자동으로 UNIQUE INDEX 생성
   - ' SELECT * FROM USER_CONSTRAINTS; ' 실행하여, 제약 조건 정보 확인시 ' INDEX_OWNER ', ' INDEX_NAME '
- 외래키가 PRIMARY KEY 참조중일시, FK 삭제후에, PK 삭제 가능
- 테이블의 컬럼이, 어느 외래키가 참조중일 경우 삭제 불가 ( FK 선삭제 후에, 해당 테이블 삭제 가능 )
```SQL
ALTER TABLE HR.EMP
ADD CONSTRAINT emp_id_pk PRIMARY KEY(id); /* ' emp_id_pk ' == 제약조건명,' id ' 컬럼에 제약 조건 설정*/


ALTER TABLE HR.EMP
ADD PRIMARY KEY(id); /* CONSTRANT_NAME 자동 생성 (제약 조건명 자동 생성) */
```
> PRIMARY KEY 생성 예제 코드

</br></br></br>
```SQL
ALTER TABLE HR.EMP
DROP PRIMARY KEY; /* PRIMARY 키는 테이블당 1개로 유일하기에, 이렇게도 삭제 가능 */


ALTER TABLE HR.DEPT
DROP CONSTRAINT dept_pk CASCADE; /* ' dept_pk ' PK 참조중인, 외래키들 삭제 +  ' dept_pk ' PK 삭제 */
```

#### FOREIGN KEY ( 외래키 )
- 참조 무결성 제약 조건
- 동일한 테이블이나, 다른 테이블의 PRIMARY KEY, UNIQUE KEY 제약 조건을 참조
- 중복 값, ' NULL ' 값 허용
- 마지막 줄에 ON 절 == 조건으로 바라보는, 기준 컬럼의 삭제시 옵션
```SQL
ALTER TABLE HR.EMP
ADD CONSTRAINT emp_dept_id_fk /* ' emp_dept_id_fk ' 이름으로 FOREIGN KEY 생성 */
FOREIGN KEY(dept_id) /* ' HR.EMP.dept_id ' 컬럼을 지정 */
REFERENCES HR.DEPT(dept_id); /* ' HR.DEPT(dept_id) ' 內 존재하는 값 또는 NULL 입력 가능하게 */
-- ON DELETE RESTRICT (기본값) == ' HR.EMP.dept_id ' 컬럼에 데이터 존재시, ' HR.DEPT.dept_id '삭제 불가
-- ON DELETE CASCADE ==  ' HR.DEPT.dept_id '삭제시, ' HR.EMP.dept_id ' 컬럼 데이터들도 삭제
-- ON DELETE SET NULL == ' HR.DEPT.dept_id '삭제시, ' HR.EMP.dept_id ' 컬럼 값들을 ' NULL '


DELETE FROM HR.DEPT WHERE dept_id = 10; /* 현재 emp_dept_id_fk 외래키가 ' ON DELETE RESTRICT (기본값) ' 생성되어 있기에, ' HR.EMP.dept_id ' == ' 10 ' 행이 존재시, DELETE 수행 불가 ERR */


DELETE FROM HR.DEPT WHERE dept_id = 120; /* 현재 emp_dept_id_fk 외래키가 ' ON DELETE RESTRICT (기본값) ' 생성되어 있지만, ' HR.EMP.dept_id ' == ' 120 ' 행이 없기에 ( 참조중인 자식 DATA 없기에 ), DELETE 수행 가능 */


DROP TABLE HR.DEPT CASCADE CONSTRAINTS PURGE; /* ' HR.DEPT ' 참조하는, 외래키들 + ' HR.DEPT ' 테이블 삭제 */
```

#### UNIQUE
- 고유한 유일값 체크 ( 중복값 불가 )
- ' NULL ' 값 허용
-  자동으로 UNIQUE INDEX 생성
   - ' SELECT * FROM USER_CONSTRAINTS; ' 실행하여, 제약 조건 정보 확인시 ' INDEX_OWNER ', ' INDEX_NAME '
```SQL
ALTER TABLE HR.DEPT ADD CONSTRAINT dept_name_uk UNIQUE(dept_name);


INSERT INTO HR.DEPT(dept_id, dept_name) VALUES(30, '총무부'); /* 이미 ' dept_name ' == ' 총무부 ' 존재하여, INSERT 수행 불가 ERR */


ALTER TABLE HR.DEPT
DROP UNIQUE(dept_name); /* UNIQUE 조건은, 테이블당 하나가 아니므로 ' dept_name ' 컬럼명 기입 필수 */
```

#### CHECK
```SQL
ALTER TABLE HR.EMP
ADD CONSTRAINT emp_sal_ck
CHECK(sal>=1000 AND sal <= 2000);
```
> ' SELECT * FROM USER_CONSTRAINTS; ' 실행하여, 제약 조건 정보 확인시, CHECK 제약 조건과 NOT NULL 제약 조건은 CONSTRAINT_TYPE == ' C ' 저장되어 있다
- TRUE 경우에만 INSERT, UPDATE 수행
- 중복 값, ' NULL ' 값 허용

#### NOT NULL
- ' NULL ' 값 불허
- 중복 값 허용
- 열 레벨로만 정의 ( 테이블 레벨식 정의시 ERR )
```SQL
ALTER TABLE HR.EMP
MODIFY name CONSTRAINT emp_name_nn NOT NULL; /* ' emp_name_nn ' == 제약 조건명 */


ALTER TABLE HR.EMP
MODIFY name NULL; /* NOT NULL 제약 조건 해제 */
```
> ' SELECT * FROM USER_CONSTRAINTS; ' 실행하여, 제약 조건 정보 확인시, CHECK 제약 조건과 NOT NULL 제약 조건은 CONSTRAINT_TYPE == ' C ' 저장되어 있다
---
</br></br></br>
```SQL
CREATE TABLE HR.EMP(
--   id NUMBER CONSTRAINT emp_id_pk PRIMARY KEY, /* 열 레벨 정의 */
   name VARCHAR2(30) CONSTRAINT emp_name_nn NOT NULL /* NOT NULL 제약 조건은, 반드시 ' 열 레벨 ' 정의 필수 */
                        CONSTRAINT emp_name_uk UNIQUE, /* 제약 조건 2개 적용 ( 열 레벨 정의 ) */ 
   sal NUMBER,
--   dept_id NUMBER CONSTRAINT emp_dept_id_fk REFERENCES HR.DEPT(dept_id),
   CONSTRAINT emp_id_pk PRIMARY KEY(id), /* 테이블 레벨 정의 */
--   CONSTRAINT emp_name_uk UNIQUE(name), /* 테이블 레벨 정의 */
   CONSTRAINT emp_sal_ck CHECK(sal BETWEEN 1000 AND 2000),
   CONSTRAINT emp_dept_id_fk FOREIGN KEY(dept_id) REFERENCES HR.DEPT(dept_id) /* 테이블 레벨 정의 */ )
TABLESPACE users;
```

### RENAME ( 테이블명 수정 ( 변경 ) )
```SQL
RENAME EMP TO EMP_NEW; /* ' RENAME ' 키워드로의 수정은 ( 변경 ) SYS 유저도 아닌, 테이블 소유자 본인만 사용 가능 */


ALTER TABLE EMP_NEW
RENAME TO EMP;


ALTER TABLE HR.EMP RENAME TO EMP_NEW; /* SYS 유저가, 테이블명 변경시 */
```

### VIEW
#### 단순 VIEW
```SQL
CREATE OR REPLACE VIEW HR.COPY_EMP
AS
SELECT employee_id, last_name || ' ' || first_name AS "name", /* 표현식 사용시, 별칭 지정 필수 */
FROM HR.EMP
WITH READ ONLY; /* ' WITH READ ONLY ' == DML 불허 */


CREATE OR REPLACE VIEW HR.COPY_EMP(name, job, mgr, dept_id)
AS
SELECT employee_id,
        last_name || ' ' || first_name,
        job_id,
        manager_id,
        department_id
FROM HR.EMP;


INSERT INTO HR.COPY_EMP(job, mgr, dept_id)
SELECT job_id, manager_id, department_id /* INSERT SubQuery */
FROM HR.EMPLOYEES
WHERE employee_id <> 202;


UPDATE HR.COPY_EMP
SET SALARY = 999
WHERE "name" = 'Taylor Winston'; /* 뷰를 통해서, 가공된 표현식으로 UPDATE 가능 */


DELETE FROM HR.COPY_EMP
WHERE "name" = 'Taylor Winston'; /* 뷰를 통해서, 가공된 표현식으로 DELETE 가능 */


DROP VIEW HR.COPY_EMP; /* VIEW 삭제 */
```
> ' SELECT * FROM USER_OBJECTS; ' 실행하여, ' OBJECT_TYPE ' 확인시 ' VIEW '
- VIEW 생성시, ' WITH READ ONLY ' == DML 불허
- ' OR REPLACE ' 사용하여 CREATE 시, 기존 VIEW 삭제하고 새로 만든다 ( 테이블에 대한, 유저들의 권한은 그대로 유지 )
- VIEW 內 볼수있는 컬럼에 대해서만 INSERT, UPDATE, DELETE 작업 가능
  - 뷰의 SELECT 문에 존재하지 않는 컬럼에는 INSERT, UPDATE, DELETE 불가 ( ' last_name || ' ' || first_name AS "name" ' 식으로, 가공된 컬럼일지라도 ' last_name ', ' first_name ' 컬럼이, 뷰의 SELECT 문에 존재하지 않는다면 INSERT, UPDATE, DELETE 불가 )
  - ```SQL
    CREATE OR REPLACE VIEW HR.COPY_EMP
    AS
    SELECT last_name || ' ' || first_name, /* 가공된 ' name ' 컬럼에는 INSERT 불가 */
        first_name /* 가공식이 아닌, ' first_name ' 컬럼을 직접 SELECT 경우에는 INSERT 가능 */
    FROM HR.EMP;

    
    INSERT INTO HR.COPY_EMP(first_name, last_name) /* ' last_name ' 컬럼은 뷰의 SELECT 문에 없기에, INSERT 불가 ERR */
    VALUES('Winston', 'Taylor');


    UPDATE HR.COPY_EMP
    SET SALARY = 999
    WHERE last_name = 'Taylor'; /* ' last_name ' 컬럼은 뷰의 SELECT 문에 없기에, UPDATE 불가 ERR */


    DELETE FROM HR.COPY_EMP
    WHERE last_name = 'Taylor'; /* ' last_name ' 컬럼은 뷰의 SELECT 문에 없기에, DELETE 불가 ERR */
    ```
- VIEW 內 볼수없는 컬럼 中 NOT NULL 제약 조건이 걸려있으며 기본값도 없는 컬럼에는, 뷰를 통해서 INSERT 불가 ( NOT NULL 제약 조건을 만족할수 없기에 )
  - ```SQL
    CREATE TABLE HR.EMP ( first_name VARCHAR(50), /* 원본 테이블 예시 */
                        last_name VARCHAR(50),
                        salary NUMBER,
                        email VARCHAR(255) NOT NULL );


    INSERT INTO HR.COPY_EMP(first_name, last_name) /* 뷰의 SELECT 문에 ' email ' 컬럼이 없는데, NOT NULL 제약 조건이 걸려 있으며 DEFAULT 값도 없으므로, 뷰를 통한 INSERT 작업 불가 ERR */
    VALUES('William', 'KW'); 
    ```

```SQL <a id="sql-view-with-check-option-constraint"></a>
CREATE OR REPLACE VIEW HR.EMP_20
AS
SELECT *
FROM HR.EMP
WHERE dept_id = 20 /* CHECK 제약 조건의, 조건식*/
WITH CHECK OPTION CONSTRAINT emp_20_ck; /* ' emp_20_ck ' 이름으로, CHECK 제약 조건 생성, 적용 */


INSERT INTO HR.EMP_20(salary, dept_id)
VALUES(9999, NULL); /* ' dept_id = 20 ' CHECK 제약 조건 위반으로, INSERT 불가 ERR */


UPDATE HR.EMP_20
SET dept_id = 10 /* CHECK 제약 조건 위반으로, UPDATE 불가 ERR */
WHERE "name" = 'Taylor Winston';
```

#### 복합 VIEW
- 하나의 테이블이 아닌, 여러 테이블을 참조하는 VIEW
- 복합 VIEW 통해서는, DML 작업 불가
```SQL
CREATE OR REPLACE VIEW HR.COPY_EMP
AS
SELECT e.employee_id, e.last_name, e.first_name, e.job_id, e.department_id, d.department_name
FROM HR.EMPLOYEES e, HR.DEPARTMENTS d
WHERE e.department_id = d.department_id;
```
